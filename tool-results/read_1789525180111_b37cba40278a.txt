   211→function addH(t: string, h: number): string {
   212→  const [hh, mm] = t.split(":").map(Number);
   213→  const total = Math.min(22 * 60, hh * 60 + mm + h * 60);
   214→  return `${String(Math.floor(total / 60)).padStart(2, "0")}:${String(total % 60).padStart(2, "0")}`;
   215→}

   217→/** Default viewing window: today now → +2h, clamped to operating hours */
   218→function defaultWindow(): TimeWindow {
   219→  const d = new Date();
   220→  const start = toMinutes(timeStr(d));
   221→  const end = start + 120;
   222→  if (end > 22 * 60) {
   223→    // too late today → tomorrow morning 08:00–10:00
   224→    const t = new Date(d.getTime() + 24 * 3600_000);
   225→    return { date: dateStr(t), startTime: "08:00", endTime: "10:00" };
   226→  }
   227→  return { date: dateStr(d), startTime: fromMinutes(start), endTime: fromMinutes(end) };
   228→}

   245→/** Deterministic PRNG — module level so the seeded world stays stable per session. */
   246→function mulberry32(seed: number) {
   247→  let a = seed >>> 0;
   248→  return () => {
   249→    a |= 0;
   250→    a = (a + 0x6d2b79f5) | 0;
   251→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   252→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   253→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   254→  };
   255→}

   283→/**
   284→ * Rich operator world: live sessions, today's completed visits, upcoming bookings,
   285→ * a week of historical check-ins (feeds the peak-hours chart) and hourly revenue.
   286→ */
   287→function seedOperatorWorld(now: number): { reservations: Reservation[]; transactions: Txn[] } {
   288→  const rnd = mulberry32(20260916);
   289→  const nowD = new Date(now);
   290→  const today = dateStr(nowD);
   291→  const dayOpen = new Date(`${today}T06:00:00`).getTime();
   292→  let t0 = Math.max(dayOpen, now - 6 * HOUR); // earliest believable event today
   293→  if (t0 > now - 10 * MIN) t0 = now - 10 * MIN; // late-night / early-morning safety clamp
   294→
   295→  const res: Reservation[] = [];
   296→  const txns: Txn[] = [];
   297→  const addTxn = (type: TxnType, amount: number, at: number, code: string) =>
   298→    txns.push({ id: uid(), type, amount, createdAt: at, note: code });
   299→
   300→  const mk = (
   301→    r: Partial<Reservation> &
   302→      Pick<Reservation, "slotId" | "slotNumber" | "date" | "startTime" | "endTime" | "status" | "driverName" | "vehiclePlate" | "vehicleName">
   303→  ): Reservation => ({
   304→    id: uid(),
   305→    code: resCode(),
   306→    type: "WALK_IN",
   307→    serviceFee: TARIFF.walkInFee,
   308→    parkingFee: 0,
   309→    overtimeFee: 0,
   310→    refundAmount: 0,
   311→    createdAt: now,
   312→    ...r,
   313→  });
   314→
   315→  // ── 9 active sessions (one overdue) ──
   316→  const actives: { off: number; hrs: number; p: number; slot: string }[] = [
   317→    { off: 295, hrs: 2, p: 0, slot: "B-03" }, // overdue ~1h55m
   318→    { off: 233, hrs: 3, p: 1, slot: "A-05" },
   319→    { off: 172, hrs: 2, p: 3, slot: "A-09" },
   320→    { off: 138, hrs: 3, p: 5, slot: "B-06" },
   321→    { off: 95, hrs: 2, p: 6, slot: "A-12" },
   322→    { off: 71, hrs: 2, p: 8, slot: "B-09" },
   323→    { off: 47, hrs: 3, p: 9, slot: "A-17" },
   324→    { off: 26, hrs: 2, p: 10, slot: "B-13" },
   325→    { off: 9, hrs: 2, p: 11, slot: "A-01" },
   326→  ];
   327→  for (const a of actives) {
   328→    const inAt = Math.max(t0, now - a.off * MIN);
   329→    const inD = new Date(inAt);
   330→    const person = OP_PEOPLE[a.p];
   331→    const r = mk({
   332→      ...opSlot(a.slot),
   333→      date: today,
   334→      startTime: timeStr(inD),
   335→      endTime: addH(timeStr(inD), a.hrs),
   336→      status: "CHECKED_IN",
   337→      driverName: person.name,
   338→      vehiclePlate: person.plate,
   339→      vehicleName: person.vehicle,
   340→      createdAt: inAt,
   341→      checkedInAt: inAt,
   342→    });
   343→    res.push(r);
   344→    addTxn("SERVICE_FEE", TARIFF.walkInFee, inAt, r.code);
   345→  }
   346→
   347→  // ── 7 completed visits earlier today (one with overtime) ──
   348→  const done: { off: number; dur: number; planned: number; p: number; slot: string; advance: boolean }[] = [
   349→    { off: 390, dur: 95, planned: 2, p: 2, slot: "B-02", advance: false },
   350→    { off: 330, dur: 55, planned: 2, p: 4, slot: "A-06", advance: true },

   357→  for (const d of done) {

   359→    const outAt = inAt + d.dur * MIN;
   360→    const inD = new Date(inAt);
   361→    const person = OP_PEOPLE[d.p];
   362→    const lateMin = Math.max(0, d.dur - d.planned * 60);
   363→    const r = mk({
   364→      ...opSlot(d.slot),
   365→      type: d.advance ? "ADVANCE" : "WALK_IN",
   366→      serviceFee: d.advance ? TARIFF.advanceFee : TARIFF.walkInFee,

   368→      startTime: timeStr(inD),
   369→      endTime: addH(timeStr(inD), d.planned),

   371→      driverName: person.name,

   373→      vehicleName: person.vehicle,
   374→      parkingFee: parkingFee(d.dur),
   375→      overtimeFee: overtimeFee(lateMin),
   376→      createdAt: d.advance ? Math.max(t0, inAt - 40 * MIN) : inAt,
   377→      checkedInAt: inAt,
   378→      checkedOutAt: outAt,
   379→    });
   380→    res.push(r);
   381→    addTxn("SERVICE_FEE", r.serviceFee, r.createdAt, r.code);
   382→    addTxn("PARKING_FEE", r.parkingFee, outAt, r.code);
   383→    if (r.overtimeFee > 0) addTxn("OVERTIME", r.overtimeFee, outAt, r.code);

   387→  const upcoming: { startOff: number; hrs: number; p: number; slot: string; createdOff: number }[] = [
   388→    { startOff: -20, hrs: 2, p: 4, slot: "A-07", createdOff: 190 },

   392→  for (const u of upcoming) {

   395→    // snap bookings into operating hours (early-morning / late-night demos)
   396→    const h = new Date(startAt).getHours();
   397→    if (h >= 20 || h < 6) {
   398→      const tm = new Date(startAt + DAY);
   399→      date = dateStr(tm);
   400→      startAt = new Date(`${date}T09:00:00`).getTime();
   401→    }
   402→    const person = OP_PEOPLE[u.p];
   403→    const r = mk({
   404→      ...opSlot(u.slot),
   405→      type: "ADVANCE",
   406→      serviceFee: TARIFF.advanceFee,
   407→      date,
   408→      startTime: timeStr(new Date(startAt)),
   409→      endTime: addH(timeStr(new Date(startAt)), u.hrs),
   410→      status: "CONFIRMED",
   411→      driverName: person.name,
   412→      vehiclePlate: person.plate,
   413→      vehicleName: person.vehicle,
   414→      createdAt: Math.max(t0, now - u.createdOff * MIN),
   415→    });
   416→    res.push(r);
   417→    addTxn("SERVICE_FEE", TARIFF.advanceFee, r.createdAt, r.code);
   418→  }
   419→
   420→  // ── a few wallet top-ups today (activity feed flavour) ──
   421→  const tops: { off: number; amount: number }[] = [
   422→    { off: 35, amount: 100000 },
   423→    { off: 130, amount: 50000 },
   424→    { off: 260, amount: 150000 },
   425→  ];
   426→  for (const tp of tops) addTxn("TOP_UP", tp.amount, Math.max(t0, now - tp.off * MIN), "");
   427→
   428→  // ── 7 days × 8 historical check-ins → peak-hours chart ──
   429→  for (let d = 1; d <= 7; d++) {
   430→    for (let k = 0; k < 8; k++) {
   431→      const ds = dateStr(new Date(now - d * DAY));
   432→      const hour = OP_HIST_HOURS[Math.floor(rnd() * OP_HIST_HOURS.length)];
   433→      const minute = Math.floor(rnd() * 55);
   434→      const inAt = new Date(`${ds}T${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}:00`).getTime();
   435→      const closeAt = new Date(`${ds}T22:00:00`).getTime();
   436→      const outAt = Math.min(inAt + (35 + Math.floor(rnd() * 230)) * MIN, closeAt);
   437→      const durMin = Math.max(15, Math.round((outAt - inAt) / MIN));
   438→      const person = OP_PEOPLE[Math.floor(rnd() * OP_PEOPLE.length)];
   439→      const slot = OP_HIST_SLOTS[Math.floor(rnd() * OP_HIST_SLOTS.length)];
   440→      res.push(
   441→        mk({
   442→          ...opSlot(slot),
   443→          date: ds,
   444→          startTime: timeStr(new Date(inAt)),
   445→          endTime: timeStr(new Date(outAt)),
   446→          status: "COMPLETED",
   447→          driverName: person.name,
   448→          vehiclePlate: person.plate,
   449→          vehicleName: person.vehicle,
   450→          parkingFee: parkingFee(durMin),
   451→          createdAt: inAt,
   452→          checkedInAt: inAt,
   453→          checkedOutAt: outAt,
   454→        })
   455→      );
   456→    }
   457→  }
   458→
   459→  txns.sort((a, b) => b.createdAt - a.createdAt);
   460→  return { reservations: res, transactions: txns };
   461→}

   693→  scanSlot: (slotNumberRaw) => {
   694→    const now = Date.now();
   695→    const { slots, reservations, walletBalance } = get();
   696→    const code = slotNumberRaw.trim().toUpperCase().replace(/^PB-?/, "");
   697→    const slot = slots.find(
   698→      (s) => s.slotNumber === code || s.slotNumber === code.replace("-", "") || `slot-${code.replace("-", "-")}` === s.id
   699→    );
   700→    if (!slot) return { ok: false as const, reason: "unknown" as const };
   701→    if (slot.status === "MAINTENANCE") return { ok: false as const, reason: "maintenance" as const };
   702→
   703→    const activeNow = reservations.filter((r) => r.status === "CHECKED_IN").length;
   704→
   705→    // Own active session on this slot → checkout (reuse checkOut for identical fee logic)
   706→    const active = reservations.find(
   707→      (r) => r.slotId === slot.id && r.status === "CHECKED_IN"
   708→    );
   709→    if (active) {
   710→      const out = get().checkOut(active.id);
   711→      return out.ok
   712→        ? { ok: true as const, kind: "checkout" as const, reservation: active }
   713→        : { ok: false as const, reason: "insufficient" as const };
   714→    }
   715→
   716→    // Own confirmed reservation on this slot → check in
   717→    const confirmed = reservations.find(
   718→      (r) => r.slotId === slot.id && r.status === "CONFIRMED"
   719→    );
   720→    if (confirmed) {
   721→      if (activeNow >= MAX_ACTIVE_PARKING)
   722→        return { ok: false as const, reason: "max_active" as const };
   723→      set((s) => ({
   724→        reservations: s.reservations.map((r) =>
   725→          r.id === confirmed.id ? { ...r, status: "CHECKED_IN" as ResStatus, checkedInAt: now } : r
   726→        ),
   727→      }));
   728→      return { ok: true as const, kind: "checkin" as const, reservation: confirmed };
   729→    }
   730→
   731→    // Any other active/confirmed overlapping right now → busy
   732→    const busy = reservations.find(
   733→      (r) =>
   734→        r.slotId === slot.id &&
   735→        ["CONFIRMED", "CHECKED_IN"].includes(r.status) &&
   736→        r.date === dateStr(new Date(now))
   737→    );
   738→    if (busy) return { ok: false as const, reason: "busy" as const };
   739→
   740→    // Free slot → walk-in session, charged at walk-in rate
   741→    if (activeNow >= MAX_ACTIVE_PARKING)
   742→      return { ok: false as const, reason: "max_active" as const };
   743→    if (walletBalance < TARIFF.walkInFee)
   744→      return { ok: false as const, reason: "insufficient" as const };
   745→
   746→    const nowD = new Date(now);
   747→    const res: Reservation = {
   748→      id: uid(),
   749→      code: resCode(),
   750→      type: "WALK_IN",
   751→      slotId: slot.id,
   752→      slotNumber: slot.slotNumber,
   753→      date: dateStr(nowD),
   754→      startTime: timeStr(nowD),
   755→      endTime: addH(timeStr(nowD), 2),
   756→      status: "CHECKED_IN",
   757→      serviceFee: TARIFF.walkInFee,
   758→      parkingFee: 0,
   759→      overtimeFee: 0,
   760→      refundAmount: 0,
   761→      vehiclePlate: get().vehicles[0]?.licensePlate ?? "B 2143 RWZ",
   762→      vehicleName: get().vehicles[0]?.model ?? "Honda Vario 160",
   763→      driverName: get().user.name,
   764→      createdAt: now,
   765→      checkedInAt: now,
   766→    };
   767→    set((s) => ({
   768→      reservations: [res, ...s.reservations],
   769→      walletBalance: s.walletBalance - TARIFF.walkInFee,
   770→      transactions: [
   771→        { id: uid(), type: "SERVICE_FEE", amount: TARIFF.walkInFee, createdAt: now, note: res.code },
   772→        ...s.transactions,
   773→      ],
   774→    }));
   775→    return { ok: true as const, kind: "walkin" as const, reservation: res };
   776→  },
