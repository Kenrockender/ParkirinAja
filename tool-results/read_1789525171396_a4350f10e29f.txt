   106→export function toMinutes(t: string): number {
   107→  const [h, m] = t.split(":").map(Number);
   108→  return h * 60 + m;
   109→}
   110→
   111→export function fromMinutes(mins: number): string {
   112→  return `${String(Math.floor(mins / 60)).padStart(2, "0")}:${String(mins % 60).padStart(2, "0")}`;
   113→}

   115→export function dateStr(d: Date): string {
   116→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   117→}

   123→export interface TimeWindow {
   124→  date: string;
   125→  startTime: string;
   126→  endTime: string;
   127→}
   128→
   129→export function overlaps(a: TimeWindow, b: TimeWindow): boolean {
   130→  if (a.date !== b.date) return false;
   131→  const aS = toMinutes(a.startTime);
   132→  const aE = toMinutes(a.endTime);
   133→  const bS = toMinutes(b.startTime);
   134→  const bE = toMinutes(b.endTime);
   135→  return aS < bE && bS < aE;
   136→}
   137→
   138→export function validateWindow(w: TimeWindow): { ok: boolean; error?: string } {
   139→  const s = toMinutes(w.startTime);
   140→  const e = toMinutes(w.endTime);
   141→  if (!w.date || !w.startTime || !w.endTime) return { ok: false, error: "incomplete" };
   142→  if (e <= s) return { ok: false, error: "end_before_start" };
   143→  if (s < TARIFF.openHour * 60) return { ok: false, error: "before_open" };
   144→  if (e > TARIFF.closeHour * 60) return { ok: false, error: "after_close" };
   145→  if ((e - s) / 60 > 12) return { ok: false, error: "too_long" };
   146→  return { ok: true };
   147→}

   149→/** Display status of a slot for a given viewing window */
   150→export function slotStatusForWindow(
   151→  slot: Slot,
   152→  reservations: Reservation[],
   153→  window: TimeWindow
   154→): SlotStatus {
   155→  if (slot.status === "MAINTENANCE") return "MAINTENANCE";
   156→  const wStart = new Date(`${window.date}T${window.startTime}:00`).getTime();
   157→  const wEnd = new Date(`${window.date}T${window.endTime}:00`).getTime();
   158→  const now = Date.now();
   159→  for (const r of reservations) {
   160→    if (r.slotId !== slot.id) continue;
   161→    if (r.status === "CHECKED_IN") {
   162→      // A parked car physically occupies its slot until it checks out —
   163→      // even when it has overstayed its booked window (overdue case).
   164→      const rStart = r.checkedInAt ?? new Date(`${r.date}T${r.startTime}:00`).getTime();
   165→      const rEnd = Math.max(new Date(`${r.date}T${r.endTime}:00`).getTime(), now);
   166→      if (wStart <= rEnd && wEnd >= rStart) return "OCCUPIED";
   167→      continue;
   168→    }
   169→    if (r.status === "CONFIRMED" && overlaps({ date: r.date, startTime: r.startTime, endTime: r.endTime }, window)) {
   170→      return "RESERVED";
   171→    }
   172→  }
   173→  return "AVAILABLE";
   174→}

   176→/** Refund policy: 100% within 10 min of creation, 50% otherwise, 0% no-show */
   177→export function refundAmount(
   178→  serviceFee: number,
   179→  startMs: number,
   180→  cancelledAt: number,
   181→  createdAt: number
   182→): number {
   183→  if (cancelledAt - createdAt <= TARIFF.fullRefundWindowMs) return serviceFee;
   184→  void startMs;
   185→  return Math.floor((serviceFee * TARIFF.refundPartialPct) / 100);
   186→}
   187→
   188→/** Parking fee by minutes parked */
   189→export function parkingFee(minutes: number): number {
   190→  const hours = Math.max(1, Math.ceil(minutes / 60));
   191→  if (hours <= TARIFF.parkFirstHours) return TARIFF.parkFirstHoursFee;
   192→  const fee =
   193→    TARIFF.parkFirstHoursFee + (hours - TARIFF.parkFirstHours) * TARIFF.parkAddHourFee;
   194→  return Math.min(fee, TARIFF.parkMaxFee);
   195→}
   196→
   197→export function overtimeFee(minutesLate: number): number {
   198→  if (minutesLate <= 0) return 0;
   199→  return Math.ceil(minutesLate / 60) * TARIFF.overtimeFeePerHour;
   200→}

   206→export function resCode(): string {
   207→  const letters = "ABCDEFGHJKLMNPQRSTUVWXYZ";
   208→  let s = "";
   209→  for (let i = 0; i < 3; i++) s += letters[Math.floor(Math.random() * letters.length)];
   210→  return `PB-${s}${String(Math.floor(1000 + Math.random() * 9000))}`;
   211→}

   801→/** Deterministic heatmap: 7 days × 07:00–21:00 */
   802→export function buildHeatmap(): number[][] {
   803→  // rows: Mon..Sun, cols: 07..21 (15 hours)
   804→  const days = [0, 1, 2, 3, 4, 5, 6];
   805→  return days.map((d) =>
   806→    Array.from({ length: 15 }, (_, h) => {
   807→      const hour = h + 7;
   808→      let v = 0.12;
   809→      if (d < 5) {
   810→        if (hour >= 7 && hour <= 10) v = 0.55 + (hour === 8 || hour === 9 ? 0.35 : 0);
   811→        if (hour >= 11 && hour <= 13) v = 0.45;
   812→        if (hour >= 16 && hour <= 18) v = 0.6 + (hour === 17 ? 0.3 : 0);
   813→        if (hour >= 19) v = 0.25;
   814→      } else {
   815→        v = hour <= 10 ? 0.15 : hour <= 15 ? 0.22 : 0.12;
   816→      }
   817→      const jitter = ((d * 31 + h * 17) % 13) / 130;
   818→      return Math.min(1, Math.max(0.05, v + jitter - 0.05));
   819→    })
   820→  );
   821→}
