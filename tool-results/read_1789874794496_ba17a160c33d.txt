   148→   148→/** Derive the live activity feed from reservations + transactions. */
   149→   149→function buildFeed(reservations: Reservation[], transactions: Txn[]): FeedEvent[] {
   150→   150→  const evts: FeedEvent[] = [];
   151→   151→  for (const r of reservations) {
   152→   152→    if (r.type === "ADVANCE")
   153→   153→      evts.push({ id: `b-${r.id}`, at: r.createdAt, kind: "booking", name: r.driverName, slot: r.slotNumber });
   154→   154→    if (r.checkedInAt)
   155→   155→      evts.push({ id: `i-${r.id}`, at: r.checkedInAt, kind: "checkin", name: r.driverName, slot: r.slotNumber });
   156→   156→    if (r.checkedOutAt)
   157→   157→      evts.push({ id: `o-${r.id}`, at: r.checkedOutAt, kind: "checkout", name: r.driverName, slot: r.slotNumber });
   158→   158→  }
   159→   159→  for (const tx of transactions)
   160→   160→    if (tx.type === "TOP_UP") evts.push({ id: `t-${tx.id}`, at: tx.createdAt, kind: "topup", name: "", amount: tx.amount });
   161→   161→  return evts.sort((a, b) => b.at - a.at).slice(0, 24);
   162→   162→}

   164→   164→/** Revenue per hour bucket (06–22) from today's fee transactions. */
   165→   165→function hourBucketsRevenue(txns: Txn[]): number[] {
   166→   166→  const buckets = new Array(DAY_END_H - DAY_START_H + 1).fill(0);
   167→   167→  for (const tx of txns) {
   168→   168→    if (tx.type === "TOP_UP" || tx.type === "REFUND") continue;
   169→   169→    const h = new Date(tx.createdAt).getHours();
   170→   170→    if (h >= DAY_START_H && h <= DAY_END_H) buckets[h - DAY_START_H] += tx.amount;
   171→   171→  }
   172→   172→  return buckets;
   173→   173→}

   175→   175→/** Check-in counts per hour bucket (06–22) over the last 7 days. */
   176→   176→function hourBucketsCheckins(reservations: Reservation[], sinceMs: number): number[] {
   177→   177→  const buckets = new Array(DAY_END_H - DAY_START_H + 1).fill(0);
   178→   178→  for (const r of reservations) {
   179→   179→    if (!r.checkedInAt || r.checkedInAt < sinceMs) continue;
   180→   180→    const h = new Date(r.checkedInAt).getHours();
   181→   181→    if (h >= DAY_START_H && h <= DAY_END_H) buckets[h - DAY_START_H]++;
   182→   182→  }
   183→   183→  return buckets;
   184→   184→}

   186→   186→/** Max simultaneous occupancy today (sweep over check-in/out events). */
   187→   187→function peakOccupancyToday(reservations: Reservation[], dayStart: number): number {
   188→   188→  const evts: { at: number; delta: number }[] = [];
   189→   189→  for (const r of reservations) {
   190→   190→    if (r.checkedInAt && r.checkedInAt >= dayStart) evts.push({ at: r.checkedInAt, delta: 1 });
   191→   191→    if (r.checkedOutAt && r.checkedOutAt >= dayStart) evts.push({ at: r.checkedOutAt, delta: -1 });
   192→   192→  }
   193→   193→  evts.sort((a, b) => a.at - b.at || a.delta - b.delta);
   194→   194→  let cur = 0;
   195→   195→  let peak = 0;
   196→   196→  for (const e of evts) {
   197→   197→    cur += e.delta;
   198→   198→    if (cur > peak) peak = cur;
   199→   199→  }
   200→   200→  return peak;
   201→   201→}
