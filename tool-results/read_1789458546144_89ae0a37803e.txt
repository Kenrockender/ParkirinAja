    45→    45→/** Window covering "right now" — operator always sees live state */
    46→    46→function nowWindow() {
    47→    47→  const d = new Date();
    48→    48→  const start = new Date(d.getTime() - 60_000);
    49→    49→  const end = new Date(d.getTime() + 60_000);
    50→    50→  return {
    51→    51→    date: dateStr(d),
    52→    52→    startTime: timeStr(start),
    53→    53→    endTime: timeStr(end),
    54→    54→  };
    55→    55→}

    81→    81→function useTicker(active: boolean): number {
    82→    82→  const [tick, setTick] = React.useState(0);
    83→    83→  React.useEffect(() => {
    84→    84→    if (!active) return;
    85→    85→    const id = setInterval(() => setTick((t) => t + 1), 30_000);
    86→    86→    return () => clearInterval(id);
    87→    87→  }, [active]);
    88→    88→  return tick;
    89→    89→}

    91→    91→function fmtDuration(ms: number): string {
    92→    92→  const mins = Math.max(0, Math.floor(ms / 60_000));
    93→    93→  const h = Math.floor(mins / 60);
    94→    94→  const m = mins % 60;
    95→    95→  return h > 0 ? `${h}j ${m}m` : `${m}m`;
    96→    96→}
