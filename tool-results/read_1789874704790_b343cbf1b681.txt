   235→function addH(t: string, h: number): string {
   236→  const [hh, mm] = t.split(":").map(Number);
   237→  const total = Math.min(22 * 60, hh * 60 + mm + h * 60);
   238→  return `${String(Math.floor(total / 60)).padStart(2, "0")}:${String(total % 60).padStart(2, "0")}`;
   239→}

   241→/** Default viewing window: today now → +2h, clamped to operating hours */
   242→function defaultWindow(): TimeWindow {
   243→  const d = new Date();
   244→  const start = toMinutes(timeStr(d));
   245→  const end = start + 120;
   246→  if (end > 22 * 60) {
   247→    // too late today → tomorrow morning 08:00–10:00
   248→    const t = new Date(d.getTime() + 24 * 3600_000);
   249→    return { date: dateStr(t), startTime: "08:00", endTime: "10:00" };
   250→  }
   251→  return { date: dateStr(d), startTime: fromMinutes(start), endTime: fromMinutes(end) };
   252→}

   269→/** Deterministic PRNG — module level so the seeded world stays stable per session. */
   270→function mulberry32(seed: number) {
   271→  let a = seed >>> 0;
   272→  return () => {
   273→    a |= 0;
   274→    a = (a + 0x6d2b79f5) | 0;
   275→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   276→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   277→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   278→  };
   279→}

   499→/** Weighted check-in hours for the 7-day peak-hours histogram (campus rhythm). */
   500→const OP_HIST_HOURS = [7, 8, 8, 9, 9, 9, 10, 11, 12, 12, 13, 13, 14, 15, 16, 16, 17, 17, 18];
   501→const OP_HIST_SLOTS = ["A-03", "A-04", "A-08", "A-11", "A-16", "A-18", "B-04", "B-07", "B-10", "B-14"];
   502→
   503→/**
   504→ * Rich operator world: live sessions, today's completed visits, upcoming bookings,
   505→ * a week of historical check-ins (feeds the peak-hours chart) and hourly revenue.
   506→ */
   507→function seedOperatorWorld(now: number): { reservations: Reservation[]; transactions: Txn[] } {
   508→  const rnd = mulberry32(20260916);
   509→  const nowD = new Date(now);
   510→  const today = dateStr(nowD);
   511→  const dayOpen = new Date(`${today}T06:00:00`).getTime();
   512→  let t0 = Math.max(dayOpen, now - 6 * HOUR); // earliest believable event today
   513→  if (t0 > now - 10 * MIN) t0 = now - 10 * MIN; // late-night / early-morning safety clamp
   514→
   515→  const res: Reservation[] = [];
   516→  const txns: Txn[] = [];
   517→  const addTxn = (type: TxnType, amount: number, at: number, code: string) =>
   518→    txns.push({ id: uid(), type, amount, createdAt: at, note: code });
   519→
   520→  const mk = (
   521→    r: Partial<Reservation> &
   522→      Pick<Reservation, "slotId" | "slotNumber" | "date" | "startTime" | "endTime" | "status" | "driverName" | "vehiclePlate" | "vehicleName">
   523→  ): Reservation => ({
   524→    id: uid(),
   525→    code: resCode(),
   526→    type: "WALK_IN",
   527→    demandTier: "NORMAL",
   528→    serviceFee: TARIFF.walkInFee,
   529→    parkingFee: 0,
   530→    overtimeFee: 0,
   531→    refundAmount: 0,
   532→    createdAt: now,
   533→    ...r,
   534→  });
   535→
   536→  // ── 9 active sessions (one overdue) ──
   537→  const actives: { off: number; hrs: number; p: number; slot: string }[] = [
   538→    { off: 295, hrs: 2, p: 0, slot: "B-03" }, // overdue ~1h55m
   539→    { off: 233, hrs: 3, p: 1, slot: "A-05" },
   540→    { off: 172, hrs: 2, p: 3, slot: "A-09" },
   541→    { off: 138, hrs: 3, p: 5, slot: "B-06" },
   542→    { off: 95, hrs: 2, p: 6, slot: "A-12" },
   543→    { off: 71, hrs: 2, p: 8, slot: "B-09" },
   544→    { off: 47, hrs: 3, p: 9, slot: "A-17" },
   545→    { off: 26, hrs: 2, p: 10, slot: "B-13" },
   546→    { off: 9, hrs: 2, p: 11, slot: "A-01" },
   547→  ];
   548→  for (const a of actives) {
   549→    const inAt = Math.max(t0, now - a.off * MIN);
   550→    const inD = new Date(inAt);
   551→    const person = OP_PEOPLE[a.p];
   552→    const r = mk({
   553→      ...opSlot(a.slot),
   554→      date: today,
   555→      startTime: timeStr(inD),
   556→      endTime: addH(timeStr(inD), a.hrs),
   557→      status: "CHECKED_IN",
   558→      driverName: person.name,
   559→      vehiclePlate: person.plate,
   560→      vehicleName: person.vehicle,
   561→      createdAt: inAt,
   562→      checkedInAt: inAt,
   563→    });
   564→    res.push(r);
   565→    addTxn("SERVICE_FEE", TARIFF.walkInFee, inAt, r.code);
   566→  }
   567→
   568→  // ── 7 completed visits earlier today (one with overtime) ──
   569→  const done: { off: number; dur: number; planned: number; p: number; slot: string; advance: boolean }[] = [
   570→    { off: 390, dur: 95, planned: 2, p: 2, slot: "B-02", advance: false },
   571→    { off: 330, dur: 55, planned: 2, p: 4, slot: "A-06", advance: true },
   572→    { off: 300, dur: 185, planned: 2, p: 7, slot: "B-08", advance: false }, // overtime
   573→    { off: 268, dur: 140, planned: 3, p: 1, slot: "A-10", advance: false },
   574→    { off: 205, dur: 45, planned: 1, p: 5, slot: "B-05", advance: false },
   575→    { off: 115, dur: 70, planned: 2, p: 10, slot: "A-13", advance: true },
   576→    { off: 62, dur: 35, planned: 1, p: 3, slot: "B-12", advance: false },
   577→  ];
   578→  for (const d of done) {
   579→    const inAt = Math.max(t0, now - d.off * MIN);
   580→    const outAt = inAt + d.dur * MIN;
   581→    const inD = new Date(inAt);
   582→    const person = OP_PEOPLE[d.p];
   583→    const lateMin = Math.max(0, d.dur - d.planned * 60);
   584→    const r = mk({
   585→      ...opSlot(d.slot),
   586→      type: d.advance ? "ADVANCE" : "WALK_IN",
   587→      serviceFee: d.advance ? TARIFF.advanceFee : TARIFF.walkInFee,
   588→      date: today,
   589→      startTime: timeStr(inD),
   590→      endTime: addH(timeStr(inD), d.planned),
   591→      status: "COMPLETED",
   592→      driverName: person.name,
   593→      vehiclePlate: person.plate,
   594→      vehicleName: person.vehicle,
   595→      parkingFee: parkingFee(d.dur),
   596→      overtimeFee: overtimeFee(lateMin),
   597→      createdAt: d.advance ? Math.max(t0, inAt - 40 * MIN) : inAt,
   598→      checkedInAt: inAt,
   599→      checkedOutAt: outAt,
   600→    });
   601→    res.push(r);
   602→    addTxn("SERVICE_FEE", r.serviceFee, r.createdAt, r.code);
   603→    addTxn("PARKING_FEE", r.parkingFee, outAt, r.code);
   604→    if (r.overtimeFee > 0) addTxn("OVERTIME", r.overtimeFee, outAt, r.code);
   605→  }
   606→
   607→  // ── today's bookings: 1 in-window (late check-in) + 2 upcoming ──
   608→  const upcoming: { startOff: number; hrs: number; p: number; slot: string; createdOff: number }[] = [
   609→    { startOff: -20, hrs: 2, p: 4, slot: "A-07", createdOff: 190 },
   610→    { startOff: 55, hrs: 2, p: 6, slot: "B-01", createdOff: 120 },
   611→    { startOff: 180, hrs: 2, p: 11, slot: "A-15", createdOff: 65 },
   612→  ];
   613→  for (const u of upcoming) {
   614→    let startAt = now + u.startOff * MIN;
   615→    let date = today;
   616→    // snap bookings into operating hours (early-morning / late-night demos)
   617→    const h = new Date(startAt).getHours();
   618→    if (h >= 20 || h < 6) {
   619→      const tm = new Date(startAt + DAY);
   620→      date = dateStr(tm);
   621→      startAt = new Date(`${date}T09:00:00`).getTime();
   622→    }
   623→    const person = OP_PEOPLE[u.p];
   624→    const r = mk({
   625→      ...opSlot(u.slot),
   626→      type: "ADVANCE",
   627→      serviceFee: TARIFF.advanceFee,
   628→      date,
   629→      startTime: timeStr(new Date(startAt)),
   630→      endTime: addH(timeStr(new Date(startAt)), u.hrs),
   631→      status: "CONFIRMED",
   632→      driverName: person.name,
   633→      vehiclePlate: person.plate,
   634→      vehicleName: person.vehicle,
   635→      createdAt: Math.max(t0, now - u.createdOff * MIN),
   636→    });
   637→    res.push(r);
   638→    addTxn("SERVICE_FEE", TARIFF.advanceFee, r.createdAt, r.code);
   639→  }
   640→
   641→  // ── a few wallet top-ups today (activity feed flavour) ──
   642→  const tops: { off: number; amount: number }[] = [
   643→    { off: 35, amount: 100000 },
   644→    { off: 130, amount: 50000 },
   645→    { off: 260, amount: 150000 },
   646→  ];
   647→  for (const tp of tops) addTxn("TOP_UP", tp.amount, Math.max(t0, now - tp.off * MIN), "");
   648→
   649→  // ── 7 days × 8 historical check-ins → peak-hours chart ──
   650→  for (let d = 1; d <= 7; d++) {
   651→    for (let k = 0; k < 8; k++) {
   652→      const ds = dateStr(new Date(now - d * DAY));
   653→      const hour = OP_HIST_HOURS[Math.floor(rnd() * OP_HIST_HOURS.length)];
   654→      const minute = Math.floor(rnd() * 55);
   655→      const inAt = new Date(`${ds}T${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}:00`).getTime();
   656→      const closeAt = new Date(`${ds}T22:00:00`).getTime();
   657→      const outAt = Math.min(inAt + (35 + Math.floor(rnd() * 230)) * MIN, closeAt);
   658→      const durMin = Math.max(15, Math.round((outAt - inAt) / MIN));
   659→      const person = OP_PEOPLE[Math.floor(rnd() * OP_PEOPLE.length)];
   660→      const slot = OP_HIST_SLOTS[Math.floor(rnd() * OP_HIST_SLOTS.length)];
   661→      res.push(
   662→        mk({
   663→          ...opSlot(slot),
   664→          date: ds,
   665→          startTime: timeStr(new Date(inAt)),
   666→          endTime: timeStr(new Date(outAt)),
   667→          status: "COMPLETED",
   668→          driverName: person.name,
   669→          vehiclePlate: person.plate,
   670→          vehicleName: person.vehicle,
   671→          parkingFee: parkingFee(durMin),
   672→          createdAt: inAt,
   673→          checkedInAt: inAt,
   674→          checkedOutAt: outAt,
   675→        })
   676→      );
   677→    }
   678→  }
   679→
   680→  txns.sort((a, b) => b.createdAt - a.createdAt);
   681→  return { reservations: res, transactions: txns };
   682→}
