   106→    medium: "text-amber-400",
   107→    high: "text-red-400",
   108→  };

   110→    low: "from-emerald-400 to-emerald-300",
   111→    medium: "from-amber-400 to-amber-300",
   112→    high: "from-red-400 to-orange-400",
   113→  };

   118→      <motion.div
   119→        initial={{ opacity: 0, y: 10 }}
   120→        animate={{ opacity: 1, y: 0 }}
   121→        transition={{ duration: 0.4 }}
   122→        className="flex items-end justify-between"
   123→      >
   124→        <div>
   125→          <p className="text-[11px] font-medium uppercase tracking-[0.16em] text-muted-foreground">
   126→            {greet}
   127→          </p>
   128→          <h2 className="font-display text-xl font-bold tracking-tight">
   129→            {user.name.split(" ")[0]} 👋
   130→          </h2>
   131→        </div>
   132→        <div className="flex shrink-0 items-center gap-1.5">
   133→          {activeCount > 0 && (
   134→            <span className="tnum inline-flex items-center gap-1.5 rounded-full border border-primary/30 bg-primary/10 px-2.5 py-1 text-[10px] font-bold text-primary">
   135→              <CarFront className="h-3 w-3" />
   136→              {activeCount}/2 {t("activeSessions")}
   137→            </span>
   138→          )}
   139→          {campus.available && (
   140→            <span className="tnum inline-flex items-center gap-1.5 rounded-full border border-border bg-card/50 px-2.5 py-1 text-[10px] font-bold text-muted-foreground">
   141→              <Gauge className="h-3 w-3 text-primary" />
   142→              {free}/{total} {t("slotsFree")}
   143→            </span>
   144→          )}
   145→        </div>
   146→      </motion.div>
   147→
   148→      {/* ── campus bar ── */}
   149→      <CampusBar campus={campus} lang={lang} onOpen={() => setPickerOpen(true)} />
   150→
   151→      {/* ── hero availability / coming soon ── */}
   152→      {campus.available ? (
   153→      <motion.section
   154→        initial={{ opacity: 0, y: 14 }}
   155→        animate={{ opacity: 1, y: 0 }}
   156→        transition={{ duration: 0.45, delay: 0.05 }}
   157→        className="glass glow-soft relative overflow-hidden rounded-3xl p-5"
   158→      >
   159→        <div
   160→          aria-hidden
   161→          className="pointer-events-none absolute -right-16 -top-16 h-48 w-48 rounded-full bg-primary/10 blur-3xl"
   162→        />
   163→        <div className="flex items-start justify-between gap-3">
   164→          <div className="min-w-0">
   165→            <p className="text-[11px] font-semibold uppercase tracking-[0.14em] text-muted-foreground">
   166→              {t("availabilityNow")}
   167→            </p>
   168→            <div className="mt-2 flex items-baseline gap-1.5">
   169→              <span className="tnum font-display text-[3.4rem] font-bold leading-none tracking-tight text-gradient-gold">
   170→                {free}
   171→              </span>
   172→              <span className="tnum text-sm font-medium text-muted-foreground">
   173→                /{total} {t("slotsFree")}
   174→              </span>
   175→            </div>
   176→          </div>
   177→          {/* capacity dial */}
   178→          <div className="relative shrink-0" aria-hidden>
   179→            <svg width="76" height="76" viewBox="0 0 76 76" className="-rotate-90">
   180→              <circle cx="38" cy="38" r="32" fill="none" stroke="currentColor" strokeWidth="7" className="text-foreground/[0.08]" />
   181→              <circle
   182→                cx="38"
   183→                cy="38"
   184→                r="32"
   185→                fill="none"
   186→                stroke="url(#dialGrad)"
   187→                strokeWidth="7"
   188→                strokeLinecap="round"
   189→                strokeDasharray={`${(pct / 100) * 201} 999`}
   190→              />
   191→              <defs>
   192→                <linearGradient id="dialGrad" x1="0" y1="0" x2="1" y2="1">
   193→                  <stop offset="0%" stopColor="#ffd60a" />
   194→                  <stop offset="100%" stopColor="#f59e0b" />
   195→                </linearGradient>
   196→              </defs>
   197→            </svg>
   198→            <span className="tnum absolute inset-0 flex items-center justify-center text-sm font-bold">
   199→              {pct}%
   200→            </span>
   201→          </div>
   202→        </div>
   203→
   204→        <div className="mt-4 flex items-center justify-between text-xs">
   205→          <span className="text-muted-foreground">{t("level")}</span>
   206→          <span className={cn("font-bold", levelTone[levelKey])}>
   207→            {t(levelKey as "low")} · {pct}%
   208→          </span>
   209→        </div>
   210→        <div className="mt-1.5 h-2 overflow-hidden rounded-full bg-white/[0.06]">
   211→          <motion.div
   212→            initial={{ width: 0 }}
   213→            animate={{ width: `${pct}%` }}
   214→            transition={{ duration: 0.9, ease: [0.22, 1, 0.36, 1], delay: 0.2 }}
   215→            className={cn("h-full rounded-full bg-gradient-to-r", barTone[levelKey])}
   216→          />
   217→        </div>
   218→      </motion.section>
   219→      ) : (
   220→        <ComingSoonHero campus={campus} lang={lang} onSeeOthers={() => setPickerOpen(true)} />
   221→      )}
   222→
   223→      {/* ── availability search (active campuses only) ── */}
   224→      {campus.available && (
   225→      <motion.section
   226→        initial={{ opacity: 0, y: 14 }}
   227→        animate={{ opacity: 1, y: 0 }}
   228→        transition={{ duration: 0.45, delay: 0.1 }}

   231→        <p className="mb-3 flex items-center gap-1.5 text-xs font-semibold">

   261→                    toMinutes(v) + 120,
   262→                    TARIFF.closeHour * 60
   263→                  );
   264→                  setWin({ ...win, startTime: v, endTime: fromMinutes(end) });
   265→                  close();
   266→                }}
   267→              />
   268→            )}
   269→          </WindowPicker>
   270→          <WindowPicker
   271→            label={t("endTime")}
   272→            display={win.endTime}
   273→            icon={<Clock className="h-3.5 w-3.5" />}
   274→          >
   275→            {(close) => (
   276→              <TimeGrid
   277→                value={win.endTime}
   278→                from={toMinutes(win.startTime) + 30}
   279→                onPick={(v) => {
   280→                  setWin({ ...win, endTime: v });
   281→                  close();
   282→                }}
   283→              />
   284→            )}
   285→          </WindowPicker>
   286→        </div>
   287→
   288→        {/* Cari Slot button — moved here */}
   289→        <button
   290→          onClick={() => setSearched(true)}
   291→          className="glow-primary mt-3 flex h-12 w-full items-center justify-center gap-2.5 rounded-2xl bg-primary text-sm font-bold text-primary-foreground transition-transform hover:scale-[1.01] active:scale-[0.98]"
   292→        >
   293→          <Search className="h-4.5 w-4.5" />
   294→          {t("searchSlots")}
   295→          <span className="tnum rounded-full bg-black/15 px-2 py-0.5 text-[10px] font-black">
   296→            {free} {t("slotsFree")}
   297→          </span>
   298→        </button>
   299→
   300→        {/* results — parking map as the result (no text lists) */}
   301→        <AnimatePresence initial={false}>
   302→          {results && (
   303→            <motion.div
   304→              key="results"
   305→              initial={{ opacity: 0, height: 0 }}
   306→              animate={{ opacity: 1, height: "auto" }}
   307→              exit={{ opacity: 0, height: 0 }}
   308→              transition={{ duration: 0.35, ease: [0.22, 1, 0.36, 1] }}
   309→              className="overflow-hidden"
   310→            >
   311→              <div className="mt-3 space-y-2.5">
   312→                {/* summary strip */}
   313→                <div className="flex items-center justify-between gap-2 rounded-2xl border border-primary/20 bg-primary/[0.04] px-3.5 py-2.5">
   314→                  <div className="min-w-0">
   315→                    <p className="text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">
   316→                      {t("searchResults")}
   317→                    </p>
   318→                    <div className="mt-0.5 flex flex-wrap items-baseline gap-x-1.5">
   319→                      <span className="tnum font-display text-2xl font-bold leading-none text-gradient-gold">
   320→                        {results.freeSlots.length}
   321→                      </span>
   322→                      <span className="tnum text-[11px] font-medium text-muted-foreground">
   323→                        {t("slotsFound")} · {fmtDateLabel(activeWin!.date, lang)} {activeWin!.startTime}–{activeWin!.endTime}
   324→                      </span>
   325→                    </div>
   326→                  </div>
   327→                  <button
   328→                    onClick={onOpenMap}
   329→                    className="inline-flex shrink-0 items-center gap-1 rounded-full border border-primary/30 bg-primary/10 px-3 py-1.5 text-[11px] font-bold text-primary transition hover:bg-primary/20 active:scale-95"
   330→                  >
   331→                    <Maximize2 className="h-3 w-3" />
   332→                    {t("fullMap")}
   333→                  </button>
   334→                </div>
   335→
   336→                {results.freeSlots.length === 0 && (
   337→                  <div className="flex items-center gap-3 rounded-xl bg-red-400/[0.07] px-3 py-3">
   338→                    <SearchX className="h-5 w-5 shrink-0 text-red-400" />
   339→                    <div>
   340→                      <p className="text-[13px] font-bold">{t("noSlotsFound")}</p>
   341→                      <p className="text-[10.5px] text-muted-foreground">{t("noSlotsFoundSub")}</p>
   342→                    </div>
   343→                  </div>
   344→                )}
   345→
   346→                {/* the parking map IS the result */}
   347→                <ParkingMap
   348→                  compact
   349→                  onSlotPress={(s) => onSlotPress(s.id, s.slotNumber)}
   350→                />
   351→                <MapLegend />
   352→                <p className="text-center text-[10px] text-muted-foreground/70">
   353→                  {t("tapSlotHint")}
   354→                </p>
   355→              </div>
   356→            </motion.div>
   357→          )}
   358→        </AnimatePresence>
   359→      </motion.section>
   360→      )}
   361→
   362→      {/* ── ads carousel ── */}
   363→      <motion.section
   364→        initial={{ opacity: 0, y: 14 }}
   365→        animate={{ opacity: 1, y: 0 }}
   366→        transition={{ duration: 0.45, delay: 0.2 }}
   367→        className="space-y-2.5"
   368→      >
   369→        <AdCarousel lang={lang} />
   370→      </motion.section>
   371→
   372→      {/* ── heatmap (Anggrek data — active campuses only) ── */}
   373→      {campus.available && <HeatmapCard lang={lang} />}
   374→
   375→      {/* ── campus picker ── */}
   376→      <CampusPickerDialog open={pickerOpen} onClose={() => setPickerOpen(false)} />
   377→    </div>
   378→  );
   379→}

   421→/** Dialog listing every campus — Coming Soon campuses stay selectable. */

   424→  const campusId = useParkir((s) => s.campusId);

   488→        </div>
   489→        <p className="pt-1 text-center text-[10px] leading-relaxed text-muted-foreground/70">

   491→        </p>

   493→    </Dialog>
   494→  );

   629→function heatColor(v: number): string {
   630→  if (v < 0.25) return `rgba(52, 211, 153, ${0.12 + v * 0.5})`;
   631→  if (v < 0.55) return `rgba(255, 214, 10, ${0.15 + v * 0.55})`;
   632→  return `rgba(248, 113, 113, ${0.25 + v * 0.6})`;
   633→}
